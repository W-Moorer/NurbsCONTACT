cmake_minimum_required(VERSION 3.23)
project(NurbsCONTACT VERSION 0.1.0 LANGUAGES CXX)

# Resolve libraries through CMake targets. Machine-wide vcpkg MSBuild
# integration can otherwise add every installed .lib to the link line.
if(CMAKE_GENERATOR MATCHES "^Visual Studio")
    list(APPEND CMAKE_VS_GLOBALS "VcpkgEnabled=false")
endif()

include(CTest)
# Optional synchronous host/GPU benchmark. The exchange helper has no CUDA link dependency.
add_library(nc_hinge_exchange SHARED EXCLUDE_FROM_ALL nurbscontact_gpu/native/hinge_exchange.cpp)
target_compile_features(nc_hinge_exchange PRIVATE cxx_std_20)

option(NC_WITH_OCCT "Build STEP import and CAD benchmarks with Open CASCADE" ON)
option(NC_WITH_CPU_MESH "Build the FCL / CGAL triangle contact backend" OFF)
option(NC_WITH_MESH_CCD "Include swept triangle CCD when building the CPU mesh backend" ON)
set(NC_CHEN_SOURCE_DIR "" CACHE PATH "Optional Chen 2024 TDIB-CCD source checkout for STEP events")

add_library(nurbscontact STATIC src/geometry.cpp src/axial.cpp src/query.cpp src/hht.cpp src/triangle_bvh.cpp
    src/contact_dynamics.cpp src/contact_tasks.cpp src/surface_impulse.cpp src/free_rigid.cpp src/tdib.cpp)
target_compile_features(nurbscontact PUBLIC cxx_std_20)
target_include_directories(nurbscontact PUBLIC include)
add_library(nc_compiled_cad STATIC src/compiled_part.cpp src/compiled_geometry.cpp src/compiled_bezier.cpp src/compiled_contact.cpp)
target_link_libraries(nc_compiled_cad PUBLIC nurbscontact)
add_executable(nc_contact_runtime apps/contact.cpp)
target_compile_definitions(nc_contact_runtime PRIVATE NC_COMPILED_ONLY)
target_link_libraries(nc_contact_runtime PRIVATE nc_compiled_cad)
if(BUILD_TESTING)
    add_executable(nc_compiled_poses benchmarks/compiled_poses.cpp)
    target_link_libraries(nc_compiled_poses PRIVATE nc_compiled_cad)
    add_executable(nc_compiled_tests tests/compiled_cad.cpp)
    target_link_libraries(nc_compiled_tests PRIVATE nc_compiled_cad)
    add_test(NAME compiled_cad_contacts COMMAND nc_compiled_tests)
endif()
find_package(OpenMP QUIET COMPONENTS CXX)
option(NC_WITH_TBB "Use oneTBB for parallel contact work when available" ON)
if(NC_WITH_TBB)
    find_package(TBB 2021 QUIET COMPONENTS tbb)
    if(TBB_FOUND)
        target_link_libraries(nurbscontact PRIVATE TBB::tbb)
        set_source_files_properties(src/contact_tasks.cpp PROPERTIES COMPILE_DEFINITIONS NC_WITH_TBB)
    endif()
endif()
if(OpenMP_CXX_FOUND)
    target_link_libraries(nurbscontact PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
    target_compile_options(nurbscontact PUBLIC /fp:strict PRIVATE /W4 /permissive-)
else()
    target_compile_options(nurbscontact PUBLIC -fno-fast-math -ffp-contract=off PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NC_WITH_CPU_MESH)
    find_package(fcl CONFIG REQUIRED)
    find_package(CGAL 6 CONFIG REQUIRED)
    add_library(nurbscontact_mesh STATIC src/cpu_mesh_io.cpp src/cpu_mesh_intersection.cpp src/cpu_mesh_scene.cpp
        src/cpu_mesh_solid.cpp src/cpu_mesh_regions.cpp src/cpu_mesh_features.cpp src/cpu_mesh_window.cpp)
    target_link_libraries(nurbscontact_mesh PUBLIC nurbscontact PRIVATE fcl CGAL::CGAL)
    if(NC_WITH_MESH_CCD)
        find_package(Eigen3 CONFIG REQUIRED)
        include(FetchContent)
        # The pinned CCD logger includes spdlog's bundled fmt headers directly.
        FetchContent_Declare(nc_spdlog
            GIT_REPOSITORY https://github.com/gabime/spdlog.git
            GIT_TAG ad0e89cbfb4d0c1ce4d097e134eb7be67baebb36)
        FetchContent_MakeAvailable(nc_spdlog)
        FetchContent_Declare(tight_inclusion
            GIT_REPOSITORY https://github.com/Continuous-Collision-Detection/Tight-Inclusion.git
            GIT_TAG 6f84001a790a9d3362b7c13d410037140d83fa2c)
        FetchContent_MakeAvailable(tight_inclusion)
        target_sources(nurbscontact_mesh PRIVATE src/cpu_mesh_ccd.cpp)
        target_compile_definitions(nurbscontact_mesh PUBLIC NC_WITH_MESH_CCD)
        target_link_libraries(nurbscontact_mesh PRIVATE tight_inclusion::tight_inclusion)
    endif()
    if(MSVC)
        target_compile_options(nurbscontact_mesh PRIVATE /bigobj)
    endif()
    add_executable(nc_mesh apps/mesh.cpp)
    target_link_libraries(nc_mesh PRIVATE nurbscontact_mesh)
    if(BUILD_TESTING)
        add_executable(nc_cpu_mesh_tests tests/cpu_mesh.cpp)
        target_link_libraries(nc_cpu_mesh_tests PRIVATE nurbscontact_mesh)
        add_test(NAME cpu_mesh_intersections COMMAND nc_cpu_mesh_tests)
    endif()
endif()

if(NC_WITH_OCCT)
    find_package(OpenCASCADE CONFIG REQUIRED COMPONENTS
        FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
    if(TARGET TKDESTEP)
        set(NC_OCCT_STEP_TARGET TKDESTEP)
    else()
        set(NC_OCCT_STEP_TARGET TKSTEP)
    endif()
    add_library(nurbscontact_occt STATIC src/occt.cpp src/occt_contact.cpp src/cad_contact_regions.cpp
        src/cad_solid_contact.cpp src/cad_patch_search.cpp src/cad_bezier_ray.cpp src/chen_2026_step.cpp src/cad_boundary_distance.cpp src/cad_local_solid.cpp)
    if(NC_CHEN_SOURCE_DIR)
        find_package(Eigen3 CONFIG REQUIRED)
        set(TDIB_SOURCE_DIR "${NC_CHEN_SOURCE_DIR}")
        include(benchmarks/chen/rotation_header.cmake)
        target_sources(nurbscontact_occt PRIVATE src/chen_step.cpp)
        target_compile_definitions(nurbscontact_occt PRIVATE NC_WITH_CHEN)
        target_include_directories(nurbscontact_occt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_include_directories(nurbscontact_occt SYSTEM PRIVATE "${NC_CHEN_SOURCE_DIR}/core")
        target_link_libraries(nurbscontact_occt PRIVATE Eigen3::Eigen)
    endif()
    target_include_directories(nurbscontact_occt SYSTEM PUBLIC ${OpenCASCADE_INCLUDE_DIR})
    target_link_libraries(nurbscontact_occt PUBLIC nurbscontact
        TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKXSBase ${NC_OCCT_STEP_TARGET} TKMesh)
    add_executable(nc_step apps/step.cpp)
    target_link_libraries(nc_step PRIVATE nurbscontact_occt)
    add_executable(nc_contact apps/contact.cpp)
    add_executable(nc_compile_part apps/compile_part.cpp)
    target_link_libraries(nc_compile_part PRIVATE nurbscontact_occt nc_compiled_cad)
    target_link_libraries(nc_contact PRIVATE nurbscontact_occt nc_compiled_cad)
    if(NC_WITH_CPU_MESH)
        target_link_libraries(nc_contact PRIVATE nurbscontact_mesh)
        target_compile_definitions(nc_contact PRIVATE NC_WITH_CPU_MESH)
    endif()
    add_executable(nc_live_lines benchmarks/aes/live_lines.cpp)
    target_link_libraries(nc_live_lines PRIVATE nurbscontact_occt)
    add_executable(nc_nurbs_baseline benchmarks/aes/nurbs_baseline.cpp)
    target_link_libraries(nc_nurbs_baseline PRIVATE nurbscontact_occt)
    add_executable(nc_bench benchmarks/run.cpp)
    target_link_libraries(nc_bench PRIVATE nurbscontact_occt)
    add_executable(nc_assembly_bench benchmarks/assembly.cpp)
    target_link_libraries(nc_assembly_bench PRIVATE nurbscontact_occt TKXCAF TKLCAF TKCDF TKMesh)
    add_executable(nc_clearance_followup benchmarks/clearance_followup.cpp)
    target_link_libraries(nc_clearance_followup PRIVATE nurbscontact_occt)
    add_executable(nc_hht benchmarks/hht/main.cpp benchmarks/hht/contact.cpp benchmarks/hht/scene.cpp)
    target_link_libraries(nc_hht PRIVATE nurbscontact_occt TKMesh)
    add_executable(nc_gears benchmarks/gears/main.cpp benchmarks/gears/flank.cpp)
    target_link_libraries(nc_gears PRIVATE nurbscontact_occt TKMesh)
    add_executable(nc_roller_bearing benchmarks/gears/bearing.cpp)
    target_link_libraries(nc_roller_bearing PRIVATE nurbscontact_occt TKMesh)
endif()

    add_executable(nc_tdib_tests tests/tdib.cpp)

    target_link_libraries(nc_surface_impulse_tests PRIVATE nurbscontact)

    target_link_libraries(nc_tests PRIVATE nurbscontact)

    add_executable(nc_hht_tests tests/hht.cpp)
    target_link_libraries(nc_hht_tests PRIVATE nurbscontact)
    add_test(NAME hht_and_triangle_bvh COMMAND nc_hht_tests)
    if(NC_WITH_OCCT)
        add_executable(nc_cad_solid_tests tests/cad_solids.cpp)

        if(NC_CHEN_SOURCE_DIR)
            add_test(NAME chen_step_events COMMAND nc_cad_solid_tests --chen)
        endif()
        add_executable(nc_step_tests tests/step.cpp)
        target_link_libraries(nc_step_tests PRIVATE nurbscontact_occt)
        add_test(NAME step_round_trip COMMAND nc_step_tests)
        if(WIN32)
            set_property(TEST step_round_trip cad_solid_contact chen_2026_step_events PROPERTY ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:${OpenCASCADE_BINARY_DIR}")
            if(NC_CHEN_SOURCE_DIR)
                set_property(TEST chen_step_events PROPERTY ENVIRONMENT_MODIFICATION
                    "PATH=path_list_prepend:${OpenCASCADE_BINARY_DIR}")
            endif()
        endif()
    endif()
endif()
